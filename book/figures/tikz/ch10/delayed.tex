% Chapter 10: the delayed window. Singles from all detectors on one time line; the opener at t
% opens the prompt window [t, t + tau] and a delayed window of the same width [t + D, t + D + tau],
% D = 100 ns = 25 tau in MiniDAQ (drawn shorter). A pair in the prompt window may be a true or a
% random; a pair in the delayed window can only be a random.
\begin{tikzpicture}[x=1.2cm,y=1cm]
  \def\tw{0.9}
  \def\dl{7.0}
  % the time line and the singles
  \draw[->,d2dmuted,line width=0.7pt] (-0.6,0) -- (10.6,0) node[note,anchor=north east] {time};
  % windows
  \fill[partcolor!14!white] (0,0) rectangle (\tw,1.6);
  \draw[partcolor,line width=0.8pt] (0,0) rectangle (\tw,1.6);
  \fill[d2dcat1!18!white] (\dl,0) rectangle (\dl+\tw,1.6);
  \draw[d2dcat1!80!d2dink,line width=0.8pt,dash pattern=on 3pt off 1.5pt] (\dl,0) rectangle (\dl+\tw,1.6);
  \node[lab,anchor=south,text=partcolor] at (\tw/2,1.65) {prompt};
  \node[lab,anchor=south,text=d2dcat1!80!d2dink] at (\dl+\tw/2,1.65) {delayed};
  % other singles, uncorrelated with the opener: gray ticks
  \foreach \x in {-0.4,1.7,2.6,3.4,4.9,5.6,6.3,8.4,9.3,10.1} {\draw[d2dmuted,line width=1pt] (\x,0) -- (\x,0.55);}
  % the opener, its true partner, a random in the prompt window, a single in the delayed window
  \draw[d2dink,line width=1.6pt] (0,0) -- (0,0.9) node[left,lab] {opener $t$};
  \draw[d2dcat0,line width=1.6pt] (0.35,0) -- (0.35,0.75);
  \draw[d2dmuted,line width=1pt] (0.7,0) -- (0.7,0.55);
  \draw[d2dmuted,line width=1pt] (\dl+0.5,0) -- (\dl+0.5,0.55);
  % brackets and dimensions
  \draw[dimline] (0,-0.35) -- (\tw,-0.35) node[midway,below,note] {$\tau$};
  \draw[dimline] (0,-1.05) -- (\dl,-1.05) node[midway,below,note] {$D$, much longer than any correlation (MiniDAQ: 100 ns $=25\tau$, drawn shorter)};
  \draw[dimline] (\dl,-0.35) -- (\dl+\tw,-0.35) node[midway,below,note] {$\tau$};
  % what each window finds
  \node[note,anchor=north west,align=left] (p) at (-0.6,3.7) {\textcolor{d2dcat0}{\textbf{true partner}}: same annihilation\\\textcolor{d2dmuted}{\textbf{other single}}: a random, if paired};
  \draw[d2dmuted,line width=0.4pt] (0.35,0.8) -- (0.35,1.3) -- (1.3,2.75);
  \node[note,anchor=north,align=center] at (\dl+\tw/2,3.7) {only uncorrelated singles can be here:\\every pair found is a random};
  \node[note,anchor=north,align=center] at (4.0,2.9) {same opener, same width,\\same rules: the delayed count\\estimates the prompt randoms};
  \draw[->,d2dmuted,line width=0.6pt] (2.9,2.3) -- (1.05,1.5);
  \draw[->,d2dmuted,line width=0.6pt] (5.1,2.3) -- (6.85,1.5);
\end{tikzpicture}
