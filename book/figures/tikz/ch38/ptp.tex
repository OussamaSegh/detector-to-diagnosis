% Chapter 38: one exchange of the Precision Time Protocol. Four time stamps, two on each clock,
% give the offset and the delay when the path takes as long both ways.
\begin{tikzpicture}[x=1cm,y=1cm]
  \draw[line width=1.2pt,draw=partcolor] (0,0) -- (0,-5.2);
  \draw[line width=1.2pt,draw=d2dcat0] (5,0) -- (5,-5.2);
  \node[title,anchor=south,text=partcolor] at (0,0.05) {master (reference time)};
  \node[title,anchor=south,text=d2dcat0] at (5,0.05) {slave (its own time)};
  \draw[flowp] (0,-0.6) -- node[lab,above,sloped] {Sync} (5,-1.6);
  \node[lab,anchor=east] at (-0.1,-0.6) {$t_1$};
  \node[lab,anchor=west] at (5.1,-1.6) {$t_2$};
  \draw[->,line width=1.3pt,draw=d2dcat0] (5,-2.8) -- node[lab,above,sloped] {Delay\_Req} (0,-3.8);
  \node[lab,anchor=west] at (5.1,-2.8) {$t_3$};
  \node[lab,anchor=east] at (-0.1,-3.8) {$t_4$};
  \draw[dimline] (5.9,-1.6) -- node[note,right,align=left] {turnaround:\\does not matter} (5.9,-2.8);
  \node[lab,anchor=west,align=left] at (8.5,-1.3) {$t_2 - t_1 = d + \theta$};
  \node[lab,anchor=west,align=left] at (8.5,-2.0) {$t_4 - t_3 = d - \theta$};
  \node[lab,anchor=west,align=left,text=partcolor] at (8.5,-2.9) {$\theta = \dfrac{(t_2 - t_1) - (t_4 - t_3)}{2}$};
  \node[lab,anchor=west,align=left,text=partcolor] at (8.5,-3.8) {$d = \dfrac{(t_2 - t_1) + (t_4 - t_3)}{2}$};
  \node[note,anchor=west,align=left] at (8.5,-4.7) {$\theta$: offset of the slave's clock; $d$: one-way delay.\\A path slower one way by $a$ errs by $a/2$.};
\end{tikzpicture}
