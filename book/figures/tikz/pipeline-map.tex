% The MiniDAQ map of the Part openers (docs/PLAN-interior-design.md, section 3.3): the data
% path from the detector to the hospital, above the foundations every stage stands on.
% \dprepmap{lit keys}{width} draws it into \dmapbox (and its height into \dmapheight) with the
% listed stages in the Part color. Keys: det ro sort lm sino corr recon qc dicom (stages),
% cpp reg test hw py team (foundations).
\newsavebox{\dmapbox}
\newlength{\dmapheight}
\newcommand{\dmapstages}{det/Detector/simulator, ro/Readout/link, sort/Coincidence/sorter, lm/List-mode/writer, sino/Sinogram/histogrammer, corr/Calibration/corrections, recon/Recon-/struction, qc/Image/quality, dicom/DICOM/archive}
\newcommand{\dmapbase}{cpp/Low-latency C++ engine, reg/Regulated file, test/Tests and CI, hw/Hardware and platform, py/Analysis and build, team/Team and tools}
\newcommand{\dprepmap}[2]{%
  \foreach \k/\a/\b in \dmapstages {\global\expandafter\let\csname dlit@\k\endcsname\relax}%
  \foreach \k/\a in \dmapbase {\global\expandafter\let\csname dlit@\k\endcsname\relax}%
  \foreach \k in {#1} {\global\expandafter\def\csname dlit@\k\endcsname{1}}%
  \global\sbox{\dmapbox}{\resizebox{#2}{!}{\begin{tikzpicture}[x=1cm,y=1cm,
      st/.style={rounded corners=3pt,minimum width=2.0cm,minimum height=1.15cm,align=center,
        font=\sffamily\bfseries\footnotesize,inner sep=1pt},
      lit/.style={fill=partcolor,draw=partcolor,text=white},
      dim/.style={fill=d2drule!30!white,draw=d2drule,text=d2dmuted},
      arr/.style={-{Stealth[length=2.2mm]},line width=0.9pt,d2dmuted}]
    \foreach \k/\a/\b [count=\i] in \dmapstages {
      \expandafter\ifx\csname dlit@\k\endcsname\relax\def\sty{dim}\else\def\sty{lit}\fi
      \node[st,\sty] (s\i) at ({(\i-1)*2.45},0) {\a\\\b};
    }
    \foreach \i [evaluate=\i as \j using int(\i+1)] in {1,...,8} {\draw[arr] (s\i) -- (s\j);}
    \node[font=\sffamily\scriptsize,text=d2dmuted,anchor=east] at (-1.15,0) {photons};
    \node[font=\sffamily\scriptsize,text=d2dmuted,anchor=west] at (20.75,0) {hospital};
    \foreach \k/\a [count=\i] in \dmapbase {
      \expandafter\ifx\csname dlit@\k\endcsname\relax\def\sty{dim}\else\def\sty{lit}\fi
      \node[st,\sty,minimum width=3.32cm,minimum height=0.75cm,font=\sffamily\bfseries\scriptsize] at ({-1.0+1.66+(\i-1)*3.62},-1.45) {\a};
    }
  \end{tikzpicture}}}%
  \global\dmapheight=\ht\dmapbox \global\advance\dmapheight by \dp\dmapbox}
