% Chapter 29: threads in the projector. Forward: each thread owns a contiguous chunk of lines and
% writes only its outputs. Back: each thread adds into its own image; the images are summed in
% thread order, so a run with the same number of threads gives the same bits.
\begin{tikzpicture}[x=1cm,y=1cm]
  \node[font=\sffamily\small\bfseries,text=partcolor,anchor=west] at (-0.2,1.0) {forward: chunks of lines};
  \foreach \t/\c in {0/d2dcat0,1/d2dcat2,2/d2dcat3,3/partcolor} {
    \fill[\c!40!white,draw=\c] (\t*1.3,0) rectangle ++(1.25,0.45);
    \node[font=\sffamily\scriptsize] at (\t*1.3+0.625,0.225) {thread \t};}
  \node[note,text=d2dink,anchor=north west,align=left] at (-0.2,-0.15) {line k * count / threads to\\(k+1) * count / threads: no remainder lost;\\each output written by one thread only};
  \begin{scope}[xshift=6.6cm]
  \node[font=\sffamily\small\bfseries,text=partcolor,anchor=west] at (-0.2,1.0) {back: an image per thread};
  \foreach \t/\c in {0/d2dcat0,1/d2dcat2,2/d2dcat3,3/partcolor} {
    \fill[\c!25!white,draw=\c] (\t*0.9,0) rectangle ++(0.7,0.7);
    \node[font=\sffamily\scriptsize] at (\t*0.9+0.35,0.35) {\t};}
  \draw[flow] (3.55,0.35) -- (4.25,0.35);
  \fill[d2dink!15!white,draw=d2dink] (4.35,0) rectangle ++(0.7,0.7);
  \node[font=\sffamily\scriptsize] at (4.7,0.35) {$\Sigma$};
  \node[note,text=d2dink,anchor=north west,align=left] at (-0.2,-0.15) {no lock, no waiting; summed in thread\\order: same bits for the same thread\\count, 1e-12 between counts};
  \end{scope}
\end{tikzpicture}
