% Chapter 8: a brain PET scanner, the front cover of the gantry removed to show the detector ring.
% A drawn view, not to scale. The ring lies in the plane of the front face; a decay in the head
% sends two photons to two blocks on opposite sides, and the line between them is the LOR.
\begin{tikzpicture}[x=1cm,y=1cm]
  % sizes of the drawing: the housing, the detector ring and the bore, as ellipses (rx, ry)
  \def\hx{1.3}\def\hy{2.6}   % housing
  \def\ox{1.02}\def\oy{2.05} % outer edge of the detector ring
  \def\bx{0.8}\def\by{1.6}   % bore
  \def\L{4.2}                % depth of the housing
  % the housing: front face, side and back
  \fill[d2drule!45!white,draw=d2dmuted,line width=0.7pt]
    (0,\hy) -- (\L,\hy) arc[start angle=90,end angle=-90,x radius=\hx,y radius=\hy]
    -- (0,-\hy) arc[start angle=270,end angle=90,x radius=\hx,y radius=\hy] -- cycle;
  % the detector ring continues along the axis, behind the cover
  \draw[partcolor,line width=0.7pt,dash pattern=on 3pt off 2pt]
    (1.6,\oy) arc[start angle=90,end angle=-90,x radius=\ox,y radius=\oy];
  \draw[partcolor,line width=0.5pt,dash pattern=on 3pt off 2pt] (0,\oy) -- (1.6,\oy) (0,-\oy) -- (1.6,-\oy);
  % the front face, and the tunnel of the bore
  \fill[d2drule!20!white,draw=d2dmuted,line width=0.7pt] (0,0) ellipse[x radius=\hx,y radius=\hy];
  \fill[partcolor!10!white] (0,0) ellipse[x radius=\ox,y radius=\oy];
  \fill[d2dink!16!white,draw=d2dmuted,line width=0.6pt] (0,0) ellipse[x radius=\bx,y radius=\by];
  \begin{scope}
    \clip (0,0) ellipse[x radius=\bx,y radius=\by];
    \fill[d2dink!28!white] (0.35,0) ellipse[x radius=0.55,y radius=1.25];
  \end{scope}
  % the detector ring: 24 blocks between the bore and the ring's outer edge
  \foreach \a in {0,15,...,345} {
    \fill[partcolor!70!white,draw=white,line width=0.3pt]
      ({\bx*cos(\a+6.5)},{\by*sin(\a+6.5)}) -- ({\ox*cos(\a+6.5)},{\oy*sin(\a+6.5)})
      arc[start angle=\a+6.5,end angle=\a-6.5,x radius=\ox,y radius=\oy]
      -- ({\bx*cos(\a-6.5)},{\by*sin(\a-6.5)})
      arc[start angle=\a-6.5,end angle=\a+6.5,x radius=\bx,y radius=\by] -- cycle;
  }
  % the two blocks that see the photons, in the second color
  \foreach \a in {75,255} {
    \fill[d2dcat1,draw=d2dink,line width=0.5pt]
      ({\bx*cos(\a+6.5)},{\by*sin(\a+6.5)}) -- ({\ox*cos(\a+6.5)},{\oy*sin(\a+6.5)})
      arc[start angle=\a+6.5,end angle=\a-6.5,x radius=\ox,y radius=\oy]
      -- ({\bx*cos(\a-6.5)},{\by*sin(\a-6.5)})
      arc[start angle=\a-6.5,end angle=\a+6.5,x radius=\bx,y radius=\by] -- cycle;
  }
  % the bed and the patient, entering the bore from the front
  \fill[d2dmuted!55!white,draw=d2dmuted,line width=0.6pt] (-5.6,-0.8) -- (2.2,-0.8) -- (2.0,-1.0) -- (-5.8,-1.0) -- cycle;
  \fill[d2dmuted!35!white] (-5.6,-1.0) rectangle (-5.35,-2.9) (-1.9,-1.0) rectangle (-1.65,-2.9);
  \fill[d2dcat4!22!white,draw=d2dcat4,line width=0.6pt]
    (-5.3,-0.8) -- (-5.3,-0.3) .. controls (-4.2,-0.05) and (-2.2,-0.05) .. (-1.05,-0.25)
    -- (-0.55,-0.35) -- (-0.55,-0.8) -- cycle;
  \fill[d2dcat4!22!white,draw=d2dcat4,line width=0.6pt] (-0.02,-0.25) ellipse[x radius=0.55,y radius=0.55];
  % a decay in the head, its two photons and the line of response
  \draw[d2dink,line width=0.5pt,dash pattern=on 2pt off 1.5pt]
    ({0.91*cos(75)},{1.83*sin(75)}) -- ({0.91*cos(255)},{1.83*sin(255)});
  \coordinate (d) at ({0.91*cos(75)*0.02},{1.83*sin(75)*0.02});
  \draw[photon,d2dcat3] (d) -- ({0.8*cos(75)},{1.58*sin(75)});
  \draw[photon,d2dcat3] (d) -- ({0.8*cos(255)},{1.58*sin(255)});
  \fill[d2dcat3] (d) circle (0.07);
  % labels on the right, with leaders
  \begin{scope}[every node/.style={lab,anchor=west}]
    \node (lh) at (6.0,2.7)  {gantry housing};
    \node (lr) at (6.0,1.85) {detector ring: 24 blocks of crystals\\and SiPMs (next figure)};
    \node (la) at (6.0,0.85) {the ring is several crystals\\deep along the axis};
    \node (ll) at (6.0,-0.25) {two 511 keV photons from one\\decay: the line of response};
    \node (lb) at (6.0,-1.35) {patient bed};
  \end{scope}
  \draw[d2dmuted,line width=0.4pt] (lh.west) -- (4.6,2.4);
  \draw[d2dmuted,line width=0.4pt] (lr.west) -- ({0.92*cos(20)},{1.84*sin(20)});
  \draw[d2dmuted,line width=0.4pt] (la.west) -- (2.45,0.85);
  \draw[d2dmuted,line width=0.4pt] (ll.west) -- ({0.5*cos(75)},{1.0*sin(75)});
  \draw[d2dmuted,line width=0.4pt] (lb.west) -- (2.1,-0.9);
  % the axial field of view
  \draw[dimline] (0,3.0) -- (1.6,3.0) node[midway,above,note] {axial field of view};
  \draw[d2drule,line width=0.4pt] (0,2.65) -- (0,3.1) (1.6,2.65) -- (1.6,3.1);
\end{tikzpicture}
