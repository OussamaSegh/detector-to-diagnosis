% Covers of Detector to Diagnosis and of its solutions manual. book/build.sh compiles this file
% with XeLaTeX in two forms, for the trim size \nlftrimw x \nlftrimh in (8.5 x 11; the front
% cover alone also for other sizes, scaled):
% - the front cover alone: page 1 of the screen PDF, and the EPUB cover as a PNG;
%     xelatex -jobname=cover-book "\def\nlfvolume{book}\def\nlfassets{book/assets}\input{cover.tex}"
% - the full-wrap paperback cover for KDP: bleed, back cover, spine and front cover, with the
%   spine width computed from the interior's page count;
%     xelatex "\def\nlfvolume{book}\def\nlfwrap{1}\def\nlfspine{1.529}\def\nlfpages{679}
%              \def\nlfauthor{Name}\def\nlfassets{book/assets}\input{cover.tex}"
% \nlfvolume is book (violet) or solutions (teal, "Solutions Manual"). \nlfauthor, when given,
% goes on the front cover and the spine.
%
% The artwork is a PET scan simulated end to end by book/tools/cover_art.py: a detector ring
% with the reconstructed brain image forming inside it out of the lines of response (front),
% and the four steps singles, coincidences, sinogram and image (back). The script writes
% cover-front-<volume>.jpg (the front panel with bleed on all four sides) and
% cover-back-<volume>.jpg (the back panel with its outer bleed) at 300 dpi; the text is set
% here, so it stays vector. The positions marked "shared with cover_art.py" must match it.
%
% KDP: bleed 0.125 in on the outer edges, text at least 0.375 in inside the trim, spine text
% only from 80 pages, and the barcode area (2 x 1.2 in, at the lower right of the back cover
% next to the spine) left empty.

\fi

% Spine, centered on its middle line. KDP keeps 0.0625 in on each side of the spine text.
\newcommand{\nlfspinetext}{%
  \pgfmathsetmacro{\fs}{min(22, 0.40*72*(\nlfspine-0.125))}%
  \pgfmathsetmacro{\fa}{min(15, 0.80*\fs)}%
  \begin{scope}[shift={(\nlfbleed+\nlftrimw+\nlfspine/2, \nlfbleed)}]
    \foreach \k in {0,...,23} {
      \pgfmathsetmacro{\aa}{15*\k}
      \pgfmathsetmacro{\mix}{100*abs(mod(\k,12)-6)/6}
      \fill[rulea!\mix!ruleb] (0,10.30) ++(\aa+3:{min(0.26,0.30*\nlfspine)}) arc[start angle=\aa+3, end angle=\aa+12, radius={min(0.26,0.30*\nlfspine)}]
        -- ++(\aa+12:{min(0.07,0.08*\nlfspine)}) arc[start angle=\aa+12, end angle=\aa+3, radius={min(0.33,0.38*\nlfspine)}] -- cycle;
    }
    \node[rotate=-90, anchor=west, inner sep=0, text=white,
          font=\bfseries\fontsize{\fs}{\fs}\selectfont]
      at (0,9.65) {Detector {\mdseries\color{accent}to} Diagnosis\ifx\nlfvolume\nlfsolutions
        {\mdseries\color{warm}\enspace Solutions Manual}\fi};
    \ifx\nlfauthor\nlfempty\else
      \node[rotate=-90, anchor=east, inner sep=0, text=subtle,
            font=\fontsize{\fa}{\fa}\selectfont]
        at (0,0.55) {\nlfauthor};
    \fi
  \end{scope}
}
