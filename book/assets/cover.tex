% Covers of Detector to Diagnosis and of its solutions manual. book/build.sh compiles this file
% with XeLaTeX in two forms, for the trim size \nlftrimw x \nlftrimh in (8.5 x 11; the front
% cover alone also for other sizes, scaled):
% - the front cover alone: page 1 of the screen PDF, and the EPUB cover as a PNG;
%     xelatex -jobname=cover-book "\def\nlfvolume{book}\def\nlfassets{book/assets}\input{cover.tex}"
% - the full-wrap paperback cover for KDP: bleed, back cover, spine and front cover, with the
%   spine width computed from the interior's page count;
%     xelatex "\def\nlfvolume{book}\def\nlfwrap{1}\def\nlfspine{1.6867}\def\nlfpages{749}
%              \def\nlfauthor{Name}\def\nlfassets{book/assets}\input{cover.tex}"
% \nlfvolume is book or solutions ("Solutions Manual" in the kicker, the subtitle and the spine).
% \nlfauthor, when given, goes on the front cover and the spine.
%
% Two designs, drawn by book/tools/cover_art.py from one simulated PET scan (a brain phantom,
% a sinogram with Poisson noise, an MLEM image) at 300 dpi. The book is Plate: the
% reconstruction over an ink band (BAND, 3.45 in). The solutions manual is Inverse: the same
% slice printed in teal and ink on paper, over a paper band (SOLUTIONS_BAND, 4.15 in). Each
% back is the field its spine uses, with the four steps singles, coincidences, sinogram and
% image. The type is set here, in EB Garamond and Inter, so it stays vector. The positions
% marked "shared with cover_art.py" must match the script.
%
% KDP: bleed 0.125 in on the outer edges, text at least 0.375 in inside the trim (this file
% keeps 0.58 in), spine text only from 80 pages, and the barcode area (2 x 1.2 in, at the
% lower right of the back cover next to the spine) left empty. The spine and the back are the
% same field, ink for the book and paper for the solutions, so they meet without a seam. The
% Inverse colors are a dark teal and a near-black on paper, which keep their order when every
% color is replaced by the gray of its lightness (the KDP black-ink profile).
\providecommand{\nlftrimw}{8.5}
\providecommand{\nlftrimh}{11}
\providecommand{\nlfwrap}{0}
\providecommand{\nlfspine}{0}
\providecommand{\nlfpages}{0}

\providecommand{\nlfassets}{.}
\providecommand{\nlfvolume}{book}

\ifnum\nlfwrap=1

  \edef\nlfpaperh{\fpeval{\nlftrimh + 2*\nlfbleed}}
  \edef\nlfbx{\nlfbleed}                                     % back trim, lower left
  \edef\nlffx{\fpeval{\nlfbleed + \nlftrimw + \nlfspine}}   % front trim, lower left
  \edef\nlfy{\nlfbleed}
\else
  \def\nlfspine{0}
  \edef\nlfpaperw{\nlftrimw}
  \edef\nlfpaperh{\nlftrimh}
  \def\nlffx{0}
  \def\nlfy{0}
\fi

% Spine, centered on its middle line. KDP keeps 0.0625 in on each side of the spine text.

  \pgfmathsetmacro{\fs}{min(24, 0.36*72*(\nlfspine-0.125))}%
  \pgfmathsetmacro{\fa}{min(12, 0.55*\fs)}%
  \pgfmathsetmacro{\fsol}{0.62*\fs}%
  \begin{scope}[shift={(\nlfbleed+\nlftrimw+\nlfspine/2, \nlfbleed)}]
    \draw[gold, line width=0.7pt] (0,10.55) -- (0,10.85);
    \node[rotate=-90, anchor=west, inner sep=0, text=cream,
          font=\fontsize{\fs}{\fs}\selectfont]
      at (0,10.15) {Detector {\itshape\color{gold}to} Diagnosis\ifx\nlfvolume\nlfsolutions
        {\mdseries\fontsize{\fsol}{\fsol}\selectfont\color{gold}\enspace Solutions Manual}\fi};

      \node[rotate=-90, anchor=east, inner sep=0, text=mute,
            font=\uifont\fontsize{\fa}{\fa}\selectfont]
        at (0,0.48) {\nlfauthor};
    \fi

  \end{scope}
